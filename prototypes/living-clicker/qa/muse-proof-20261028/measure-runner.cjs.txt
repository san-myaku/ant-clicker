function instrument(s){
 const scopes=[],pose={},sweep={},movement={calls:0,translated:0,totalDistance:0,maxDistance:0,distances:{zero:0,under025:0,under075:0,under125:0,under175:0,other:0}};
 for(const name of ['move','retreat','surfacePose','path','planPath','storageDrop','storageDestination','plan','homeSpots']){
  const fn=s[name];if(!fn)continue;s[name]=function(...args){scopes.push(name);const old=name==='move'?{x:args[0].x,y:args[0].y}:null;
   try{return fn.apply(this,args);}finally{if(old){const d=Math.hypot(args[0].x-old.x,args[0].y-old.y);movement.calls++;if(d>0){movement.translated++;movement.totalDistance+=d;movement.maxDistance=Math.max(movement.maxDistance,d);}movement.distances[d===0?'zero':d<.025?'under025':d<.075?'under075':d<.125?'under125':d<.175?'under175':'other']++;}scopes.pop();}};
 }
 const p=s.poseClear;s.poseClear=function(point,kind=null,ignore=-1,margin=.045){const key=(scopes.at(-1)||'other')+' / '+margin;pose[key]=(pose[key]||0)+1;return p.call(this,point,kind,ignore,margin);};

 return {pose,sweep,movement};
}
