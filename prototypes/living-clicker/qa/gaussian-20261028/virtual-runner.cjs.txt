   for(const b of under){const shape=G.Physics.SHAPES[b.kind],top=b.y-shape.hy;if(Math.abs(x-b.x)<hx+shape.hx-1e-7&&top>=y+hy-1e-7)landing=Math.min(landing,top-hy);}
   if(this.grainRoom[G.GRAIN.at(x,landing)]!==rid||this.physics.terrainOverlap({kind,x,y:landing})||this.physics.penetration({kind,x,y:landing})>1e-7)continue;
   stats.placements++;const offset=(landing-y)*.03;let density=0,firstCross=0;
   for(let j=0;j<stock.length;j++){const b=stock[j];density+=Math.exp(-((x-b.x)**2+(landing-b.y)**2)/1.5);stats.fullTerms++;if(best&&!firstCross&&density+offset>best.score)firstCross=j+1;}
   if(firstCross)for(const n of [1,4,8,16,32]){const stop=Math.min(stock.length,Math.ceil(firstCross/n)*n);stats.skippedTerms[n]+=stock.length-stop;if(stop<stock.length)stats.pruned[n]++;}
   const choice={angle,x,y,landing,score:density+offset};if(!best||choice.score<best.score||choice.score===best.score&&choice.angle<best.angle)best=choice;
  }
  const result=original.call(this,rid,point,kind,actualReach,inputs);assert.deepEqual(best,result,'仮想計算も元の完全評価と一致');return result;
 };return stats;
}
